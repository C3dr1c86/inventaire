# INSTRUCTIONS :
# 1. Ouvrez l'interface web du NAS : http://192.168.1.37
# 2. Ouvrez Container Manager → Terminal (ou n'importe quel conteneur → Terminal)
# 3. Copiez-collez TOUTES les commandes ci-dessous en UNE SEULE FOIS
# 4. Appuyez sur Entrée et attendez (3-5 minutes)

# ============================================
# COMMANDES À COPIER-COLLER
# ============================================

cd /tmp && \
rm -rf inventaire-main inventaire.tar.gz && \
wget -q https://github.com/C3dr1c86/inventaire/archive/refs/heads/main.tar.gz -O inventaire.tar.gz && \
tar -xzf inventaire.tar.gz && \
sudo mkdir -p /home/Cédric/inventaire && \
sudo cp -r inventaire-main/* /home/Cédric/inventaire/ && \
cd /home/Cédric/inventaire && \
echo "✅ Code téléchargé" && \
echo "🔨 Build frontend (3-5 min)..." && \
sudo docker build --no-cache -t ghcr.io/c3dr1c86/inventaire/frontend:latest -f frontend/Dockerfile frontend && \
echo "✅ Build terminé" && \
sudo mkdir -p /home/Cédric/docker/inventaire/data/{database,exports,uploads} && \
sudo cp deployment/nas/docker-compose-tailscale-https.yml /home/Cédric/docker/inventaire/docker-compose.yml && \
sudo cp deployment/nas/nginx-tailscale.conf /home/Cédric/docker/inventaire/ && \
echo "✅ Config copiée" && \
cd /home/Cédric/docker/inventaire && \
sudo docker compose down 2>/dev/null || true && \
sudo docker compose up -d && \
echo "✅ Conteneurs démarrés" && \
sleep 35 && \
sudo docker exec inventaire-tailscale tailscale serve --bg --https=443 http://localhost:80 2>/dev/null || echo "✅ HTTPS déjà configuré" && \
echo "" && \
echo "========================================" && \
echo "  ✅ DÉPLOIEMENT TERMINÉ !" && \
echo "========================================" && \
echo "" && \
echo "📱 Accès : https://inventaire-nas.tail53f399.ts.net" && \
echo "🔄 Testez avec Ctrl+Shift+R dans le navigateur"

# ============================================
# FIN DU SCRIPT
